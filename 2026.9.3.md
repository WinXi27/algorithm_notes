# 209. 长度最小的子数组
# 滑动窗口 解法
```cpp
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int sum=0;
        int l=0;
        int minn=nums.size()+1;
        for(int r=0;r<nums.size();++r){
            sum+=nums[r];
            if(sum>=target){
                while(sum>=target){
                    sum-=nums[l++];
                }
                minn=min(minn,r-l+1+1);
            }else continue;
        }
        return minn==nums.size()+1?0:minn;
    }
};
```
