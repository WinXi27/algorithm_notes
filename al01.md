# 1290. 二进制链表转整数
```cpp
class Solution {
public:
    int getDecimalValue(ListNode* head) {
        int ans=0;
        for(ListNode*cur=head;cur;cur=cur->next){
            ans=ans*2+cur->val;
        }
        return ans;
    }
};
```

# 2058. 找出临界点之间的最小和最大距离
```cpp
class Solution {
public:
    vector<int> nodesBetweenCriticalPoints(ListNode* head) {
        if(head==nullptr||head->next==nullptr||head->next->next==nullptr)return {-1,-1};
        ListNode*pre=head;
        vector<int>nums;
        int idx=0;
        for(ListNode*cur=head->next;cur;cur=cur->next){
            ListNode*Next=cur->next;
            idx++;
            if(Next==nullptr)break;
            //pre cur Next;
            if(cur->val>pre->val&&cur->val>Next->val
            ||cur->val<pre->val&&cur->val<Next->val){
                nums.push_back(idx);
            }
            pre=cur;
        }
        int minn=INT_MAX;
        for(int i=1;i<nums.size();++i){
            minn=min(minn,nums[i]-nums[i-1]);
        }
        if(nums.size()<2)return {-1,-1};
        return {minn,nums.back()-nums[0]};
    }
};
```
# 2181. 合并零之间的节点
```cpp
class Solution {
public:
    ListNode* mergeNodes(ListNode* head) {
        ListNode*dummy=new ListNode();
        ListNode*tail=dummy;
        int sum=0;
        if(!head||!head->next||!head->next->next)return nullptr;
        ListNode*cur=head;
        while(cur){
            while(cur->val==0){
                cur=cur->next;
                if(cur==nullptr)break;
            }
            while(cur->val!=0){
                sum+=cur->val;
                cur=cur->next;
            }
            tail->next=new ListNode(sum);
            tail=tail->next;
            sum=0;
            cur=cur->next;
        }
        return dummy->next;
    }
};
```
# 725. 分隔链表
## 解法1
```cpp
class Solution {
public:
    int fun(ListNode*head){
        int cnt=0;
        for(ListNode*cur=head;cur;cur=cur->next)cnt++;
        return cnt;
    }
    vector<ListNode*> splitListToParts(ListNode* head, int k) {
        vector<ListNode*>ans;
        ListNode*cur=head;
        int cnt=fun(head);
        int remind=cnt%k;
        while(cur){
            int temp=cnt/k+(remind>0?1:0);
            ListNode*dummy=new ListNode();
            ListNode*tail=dummy;
            while(temp--){
                tail->next=cur;
                tail=tail->next;
                cur=cur->next;
            }
            tail->next=nullptr;
            remind--;
            ans.push_back(dummy->next);
        }
        while(ans.size()<k){
            ans.push_back(nullptr);
        }
        return ans;
    }
};
```
## 解法2
```cpp
class Solution {
public:
    vector<ListNode*> splitListToParts(ListNode* head, int k) {
        vector<ListNode*>ans;
        int cnt=0;
        for(ListNode*cur=head;cur;cur=cur->next)cnt++;
        ListNode*p=head;
        int remind=cnt%k;
        while(p){
            int temp=cnt/k+(remind>0?1:0);
            remind--;
            ans.push_back(p);
            if(temp>0){
                while(--temp)p=p->next;
                ListNode*Nxt=p->next;
                p->next=nullptr;
                p=Nxt;
            }else break;
        }
        while(ans.size()<k)ans.push_back(nullptr);
        return ans;
    }
};
```
